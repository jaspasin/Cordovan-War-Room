    async function initApp() {

      await loadData();

      renderData();

    async function loadData() {

      if (FIREBASE_DB_URL.includes("https://console.firebase.google.com/u/0/project/cordovan-war-room/database/cordovan-war-room-default-rtdb/data/~2F")) {

        const local = localStorage.getItem('cordovan_dashboard_data');

        appData = local ? JSON.parse(local) : SEED_DATA;

      } else {

        try {

          const res = await fetch(FIREBASE_DB_URL);

          const data = await res.json();

          if (data) {

            appData = data;

          } else {

            await syncToFirebase(SEED_DATA);

            appData = SEED_DATA;

          }

        } catch (e) {

          console.error("Sync fetch error:", e);

        }

      }

      renderData();

    }

    async function syncToFirebase(data) {

      if (FIREBASE_DB_URL.includes("https://console.firebase.google.com/u/0/project/cordovan-war-room/database/cordovan-war-room-default-rtdb/data/~2F")) {

        localStorage.setItem('cordovan_dashboard_data', JSON.stringify(data));

      } else {

        try {

          await fetch(FIREBASE_DB_URL, {

            method: 'PUT',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify(data)

          });

        } catch (e) {

          console.error("Save error:", e);

        }

      }

    }

    async function updateStatus(id, newStatus) {

      const item = appData.actions.find(a => a.id === id);

      if (item) {

        item.status = newStatus;

        renderData();

        await syncToFirebase(appData);

      }

    }

    async function resetToSeed() {

      if (confirm("Reset all tasks back to default template?")) {

        appData = JSON.parse(JSON.stringify(SEED_DATA));

        await syncToFirebase(appData);

        renderData();

      }

    }

    function populateOwnerFilter() {

      const owners = [...new Set(SEED_DATA.actions.map(a => a.owner).filter(Boolean))];

      const select = document.getElementById('ownerFilter');

      owners.forEach(o => {

        const opt = document.createElement('option');

        opt.value = o;

        opt.textContent = o;

        select.appendChild(opt);

      });

    }

    function renderData() {

      const q = document.getElementById('search').value.toLowerCase();

      const st = document.getElementById('statusFilter').value;

      const ow = document.getElementById('ownerFilter').value;

      const filtered = (appData.actions || []).filter(a => {

        const matchQ = (a.desc + " " + a.cat + " " + a.owner + " " + a.id).toLowerCase().includes(q);

        const matchSt = st ? a.status === st : true;

        const matchOw = ow ? a.owner === ow : true;

        return matchQ && matchSt && matchOw;

      });

      // Update KPIs

      const all = appData.actions || [];
